Content-Signal: ai-train=no, search=yes, ai-input=no

# 2. Bloquer les robots d'IA spécifiques (Protection contre l'entraînement)
User-agent: GPTBot

User-agent: OAI-SearchBot
Disallow: /

User-agent: Claude-Web
Disallow: /

User-agent: ClaudeBot
Disallow: /

User-agent: Google-Extended
Disallow: /

User-agent: anthropic-ai

# 3. Autoriser Googlebot-Image à accéder aux favicons (essentiel pour l'affichage dans les résultats)
User-agent: Googlebot-Image
Allow: /favicon.ico
Allow: /assets/favicons/
Disallow: /

# 4. Règles générales
User-agent: *
# Autoriser les favicons (règles Allow plus spécifiques que les Disallow génériques)
Allow: /favicon.ico
Allow: /assets/favicons/
# Interdire le scan direct des autres fichiers images (hors favicons)
Disallow: /*.jpg$
Disallow: /*.jpeg$
Disallow: /*.png$
Disallow: /*.gif$
Disallow: /*.webp$
Disallow: /*.svg$
# Autoriser tout le reste du site
Allow: /
